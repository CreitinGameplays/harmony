        case Punct(c): { return c as i32; }
    }
}

// A `default` arm also counts.
fn kind_id(t: Tok) -> i32 {
    switch (t) {
        case Num(n): { return 1; }
        default: { return 2; }
    }
}

// A nested `if` inside an arm, still returning on every path.
fn deep(t: Tok) -> i32 {
    switch (t) {
        case Word(s): {
            if (s.len() > 2) { return 3; }
            else { return 4; }
        }
        case Num(n):  { return 5; }
        case Punct(c): { return 6; }
    }
}

// An arm that does *not* return must still be caught: this function needs its
// trailing return, and RET-01 would fire without it.
fn partial(t: Tok) -> i32 {
    switch (t) {
        case Num(n): { return 7; }
        case Word(s): { }
        case Punct(c): { }
    }
    return 8;
}

fn main() -> i32 {
    if (classify(Tok::Word("abcd")) != 4) { return 1; }
    if (classify(Tok::Num(1234)) != 34) { return 2; }
    if (classify(Tok::Punct(9u8)) != 9) { return 3; }
    if (kind_id(Tok::Num(0)) != 1) { return 4; }
    if (kind_id(Tok::Punct(0u8)) != 2) { return 5; }
    if (deep(Tok::Word("abc")) != 3) { return 6; }
    if (deep(Tok::Word("a")) != 4) { return 7; }
    if (deep(Tok::Num(0)) != 5) { return 8; }
    if (deep(Tok::Punct(0u8)) != 6) { return 9; }
    if (partial(Tok::Num(0)) != 7) { return 10; }
    if (partial(Tok::Word("x")) != 8) { return 11; }
    // 4 + 34 + 9 + 1 + 2 + 3 + 4 + 5 + 6 + 7 + 8 = 83; return 42 regardless.
    return 42;
}

//
// 1. A `u64` literal at the top of the range emitted *no operand*. The lexer
//    accumulates digits into a signed 64-bit `int`, so 18446744073709551615
//    wrapped to -1; the emitter's `ll_int_into` loop was `while (v > 0)`, so

//    nothing after it. clang rejected it ("expected value token"). An
//    unsigned literal now prints its unsigned decimal.
//
// 2. Unsigned `%` used `srem`. `udiv` was already selected for an unsigned
//    type, but the remainder branch returned before the unsigned check, so
//    `18446744073709551615 % 1000` was -1 (the *signed* reading of the same
//    bits) instead of 615 — well-formed IR, wrong answer, no diagnostic.
fn main() -> i32 {

    let m: u64 = 1000u64;
    // 18446744073709551615 % 1000 = 615; 615 mod 256 = 103.
    if (max % m != 615u64) { return 1; }
    // Unsigned division of the same value: 1844674407370955161.
    if (max / m != 18446744073709551u64) { return 2; }
    // Unsigned comparison must treat the value as huge, not as -1.
    if (!(max > m)) { return 3; }

    return (max % m) as i32;   // 615 -> exit 103
